// tablet-overlay.js - Floating Drag, Drop, & Resize OS Window Controller for AEROBRIEF export function initializeFloatingTablet(tabletId, dragHeaderId) { const tablet = document.getElementById(tabletId); const header = document.getElementById(dragHeaderId); const resizeHandle = document.getElementById("tabletResizeHandle"); const appIcon = document.getElementById("efbAppIcon"); if (!tablet || !header || !resizeHandle) return; let posX = 0, posY = 0, mouseX = 0, mouseY = 0; let isMaximized = false; let savedDimensions = { top: "150px", left: "360px", width: "320px", height: "240px" }; // ========================================== // MODULE FLOW 1: DRAG & POSITION CHANNELS // ========================================== header.onmousedown = dragMouseDown; function dragMouseDown(e) { if (isMaximized) return; // Prevent dragging while expanded full-screen e.preventDefault(); mouseX = e.clientX; mouseY = e.clientY; document.onmouseup = closeDragElement; document.onmousemove = elementDrag; } function elementDrag(e) { e.preventDefault(); posX = mouseX - e.clientX; posY = mouseY - e.clientY; mouseX = e.clientX; mouseY = e.clientY; tablet.style.top = (tablet.offsetTop - posY) + "px"; tablet.style.left = (tablet.offsetLeft - posX) + "px"; } function closeDragElement() { document.onmouseup = null; document.onmousemove = null; } // ========================================== // MODULE FLOW 2: RESIZE WINDOW MATRIX // ========================================== resizeHandle.onmousedown = initResize; function initResize(e) { e.preventDefault(); mouseX = e.clientX; mouseY = e.clientY; document.onmouseup = stopResize; document.onmousemove = resizeDrag; } function resizeDrag(e) { e.preventDefault(); const widthDelta = e.clientX - mouseX; const heightDelta = e.clientY - mouseY; mouseX = e.clientX; mouseY = e.clientY; const currentWidth = parseInt(document.defaultView.getComputedStyle(tablet).width); const currentHeight = parseInt(document.defaultView.getComputedStyle(tablet).height); tablet.style.width = (currentWidth + widthDelta) + "px"; tablet.style.height = (currentHeight + heightDelta) + "px"; } function stopResize() { document.onmouseup = null; document.onmousemove = null; } // ========================================== // MODULE FLOW 3: WINDOW OPERATIONS CONTROLS // ========================================== window.toggleWindowOperation = function(action) { if (action === 'MINIMIZE') { tablet.style.display = 'none'; if (appIcon) appIcon.style.display = 'flex'; console.log("[EFB WINDOW] App window minimized to background task tray."); } else if (action === 'RESTORE') { tablet.style.display = 'flex'; if (appIcon) appIcon.style.display = 'none'; } else if (action === 'CLOSE') { tablet.style.display = 'none'; if (appIcon) appIcon.style.display = 'none'; } else if (action === 'MAXIMIZE') { if (!isMaximized) { // Archive dimensions before maximizing savedDimensions = { top: tablet.style.top, left: tablet.style.left, width: tablet.style.width, height: tablet.style.height }; tablet.style.top = "50px"; tablet.style.left = "0px"; tablet.style.width = "100vw"; tablet.style.height = "calc(100vh - 50px)"; isMaximized = true; } else { // Restore previous small window parameters tablet.style.top = savedDimensions.top; tablet.style.left = savedDimensions.left; tablet.style.width = savedDimensions.width; tablet.style.height = savedDimensions.height; isMaximized = false; } } }; }
